= RACI
:page-summary: The BYOC RACI matrix listing every operational activity and its Responsible, Accountable, Consulted, and Informed assignments across the Customer and Akka, with descriptions and cadences.
:page-persona: operator-platform, operator-sre

include::ROOT:partial$include.adoc[]

This page lists every activity in the BYOC (Bring Your Own Cloud) deployment model, where Akka's automated operations run inside the Customer's own cloud account. Each activity is split across the four RACI roles: Responsible (who does the work), Accountable (who owns the outcome), Consulted (who provides input), and Informed (who is kept up to date).

Each role cell uses one of the following values: `C` = Customer, `A` = Akka, `±` = Akka and Customer, blank = not applicable.

Filter by category or owner, or search across activity, description, and cadence. Owner is derived from the Responsible column: Customer, Akka, or shared.

const RACI =

(function(){
  var CATS=['General','Platform','Federation Plane','Services'];
  var OWN=[['all','All'],['customer','Customer'],['akka','Akka'],['shared','Shared']];
  var ownerOf=function(r){return r==='C'?'customer':r==='A'?'akka':r==='±'?'shared':'';};
  var catSel='all',ownSel='all',q='';
  var catEl=document.getElementById('rc-cat'),ownEl=document.getElementById('rc-own'),
      qEl=document.getElementById('rc-q'),outEl=document.getElementById('rc-out'),cntEl=document.getElementById('rc-count');
  function esc(s){return String(s==null?'':s).replace(/[&<>"]/g,function(c){return {'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;'}[c];});}
  function roleCell(v){var t=(v||'').trim();if(!t)return '<td class="rc-role"></td>';var cls=t==='C'?'rc-C':t==='A'?'rc-A':'rc-S';return '<td class="rc-role"><span class="'+cls+'">'+esc(t)+'</span></td>';}
  function matches(e){
    if(catSel!=='all'&&e.cat!==catSel)return false;
    if(ownSel!=='all'&&ownerOf(e.r)!==ownSel)return false;
    if(q){var h=(e.act+' '+e.sub+' '+e.desc+' '+e.cad).toLowerCase();if(h.indexOf(q)<0)return false;}
    return true;
  }
  function pill(text,active){return '<button type="button" class="rc-pill'+(active?' on':'')+'">'+esc(text)+'</button>';}
  function renderPills(){
    catEl.innerHTML=pill('All',catSel==='all')+CATS.map(function(c){return pill(c,catSel===c);}).join('');
    ownEl.innerHTML=OWN.map(function(o){return pill(o[1],ownSel===o[0]);}).join('');
    var cb=catEl.querySelectorAll('.rc-pill');
    cb[0].onclick=function(){catSel='all';render();};
    CATS.forEach(function(c,i){cb[i+1].onclick=function(){catSel=c;render();};});
    var ob=ownEl.querySelectorAll('.rc-pill');
    OWN.forEach(function(o,i){ob[i].onclick=function(){ownSel=o[0];render();};});
  }
  function render(){
    renderPills();
    var shown=RACI.filter(matches),total=RACI.length;
    cntEl.textContent='Showing '+shown.length+' of '+total+' activities';
    if(!shown.length){outEl.innerHTML='<div class="rc-empty">No activities match the current filters.</div>';return;}
    var html='';
    CATS.forEach(function(cat){
      var rows=shown.filter(function(e){return e.cat===cat;});
      if(!rows.length)return;
      html+='<h3 class="rc-cat">'+esc(cat)+'</h3><div class="rc-scroll"><table class="rc-table">'+
        '<thead><tr><th>Sub-category</th><th>Activity</th>'+
        '<th class="rc-role" title="Responsible">R</th><th class="rc-role" title="Accountable">A</th>'+
        '<th class="rc-role" title="Consulted">C</th><th class="rc-role" title="Informed">I</th>'+
        '<th>Description</th><th>Cadence</th></tr></thead><tbody>';
      rows.forEach(function(e){
        html+='<tr><td class="rc-sub">'+esc(e.sub)+'</td><td class="rc-act">'+esc(e.act)+'</td>'+
          roleCell(e.r)+roleCell(e.a)+roleCell(e.c)+roleCell(e.i)+
          '<td class="rc-desc">'+esc(e.desc)+'</td><td class="rc-cad">'+esc(e.cad)+'</td></tr>';
      });
      html+='</tbody></table></div>';
    });
    outEl.innerHTML=html;
  }
  qEl.oninput=function(){q=qEl.value.trim().toLowerCase();render();};
  render();
})();
</script>
